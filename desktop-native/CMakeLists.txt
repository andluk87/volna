cmake_minimum_required(VERSION 3.25)
project(VolnaDesktop VERSION 0.1.3 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTORCC ON)
find_package(Qt6 6.8 REQUIRED COMPONENTS Widgets Network WebEngineWidgets WebChannel Test)
set(VOLNA_SERVER "https://volna.lknet.ru" CACHE STRING "Volna server HTTPS origin")
add_library(volna_native_core STATIC src/api.cpp src/network_error.cpp src/session_store.cpp src/sse.cpp src/theme.cpp third_party/qrcodegen/qrcodegen.cpp resources.qrc)
target_include_directories(volna_native_core PUBLIC src third_party/qrcodegen)
target_link_libraries(volna_native_core PUBLIC Qt6::Widgets Qt6::Network)
target_compile_definitions(volna_native_core PUBLIC VOLNA_NATIVE_VERSION="${PROJECT_VERSION}" VOLNA_DEFAULT_SERVER="${VOLNA_SERVER}")
if(WIN32)
 target_link_libraries(volna_native_core PRIVATE crypt32)
endif()
add_library(volna_native_ui STATIC src/window.cpp src/window_shell.cpp src/calls.cpp src/chat_delegate.cpp src/widgets.cpp src/image_viewer.cpp)
target_link_libraries(volna_native_ui PUBLIC volna_native_core Qt6::WebEngineWidgets Qt6::WebChannel)
qt_add_executable(VolnaDesktop WIN32 src/main.cpp)
target_link_libraries(VolnaDesktop PRIVATE volna_native_ui)
if(WIN32)
 set(VOLNA_ICON "${CMAKE_CURRENT_SOURCE_DIR}/../client/electron/icon.ico")
 configure_file(app.rc "${CMAKE_CURRENT_BINARY_DIR}/volna.rc" @ONLY)
 target_sources(VolnaDesktop PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/volna.rc")
endif()
if(MSVC)
 target_compile_options(volna_native_core PRIVATE /utf-8)
 target_compile_options(volna_native_ui PRIVATE /utf-8)
 target_compile_options(VolnaDesktop PRIVATE /utf-8)
endif()
include(CTest)
if(BUILD_TESTING)
 qt_add_executable(volna_native_tests tests/test_native.cpp)
 target_link_libraries(volna_native_tests PRIVATE volna_native_ui Qt6::Test)
 if(MSVC)
  target_compile_options(volna_native_tests PRIVATE /utf-8)
 endif()
 qt_add_executable(volna_native_integration tests/test_integration.cpp)
 target_link_libraries(volna_native_integration PRIVATE volna_native_ui Qt6::Test)
 if(MSVC)
  target_compile_options(volna_native_integration PRIVATE /utf-8)
 endif()
 add_test(NAME native COMMAND volna_native_tests)
 set_tests_properties(native PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
endif()
install(TARGETS VolnaDesktop RUNTIME DESTINATION . BUNDLE DESTINATION .)
install(FILES README.md third_party/telegramdesktop/LICENSE third_party/telegramdesktop/LEGAL third_party/sources.json DESTINATION licenses)
install(FILES third_party/qrcodegen/qrcodegen.hpp DESTINATION licenses/qrcodegen)
